cmake_minimum_required(VERSION 3.21)
project(JumeFiles VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Concurrent DBus MultimediaWidgets PdfWidgets Test)
find_package(SDL2 REQUIRED)
find_package(LibArchive REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(QR REQUIRED IMPORTED_TARGET libqrencode)
qt_standard_project_setup()
add_library(files-core operations.cpp operations.h zip.cpp storage.cpp storage.h)
target_link_libraries(files-core PUBLIC Qt6::Widgets Qt6::Concurrent Qt6::DBus LibArchive::LibArchive)
qt_add_executable(jume-files main.cpp window.cpp window.h preview.cpp preview.h transfer.cpp transfer.h ../gaming-shell/input.cpp)
target_include_directories(jume-files PRIVATE ../gaming-shell)
target_link_libraries(jume-files PRIVATE files-core Qt6::MultimediaWidgets Qt6::PdfWidgets SDL2::SDL2 PkgConfig::QR)
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(files-check check.cpp preview.cpp preview.h window.cpp window.h transfer.cpp transfer.h)
    target_link_libraries(files-check PRIVATE files-core Qt6::Test Qt6::MultimediaWidgets Qt6::PdfWidgets PkgConfig::QR)
    add_test(NAME files-check COMMAND files-check)
endif()
install(TARGETS jume-files RUNTIME DESTINATION bin)
